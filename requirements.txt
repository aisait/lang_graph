# requirements.txt

# - ISO/IEC/IEEE 12207:2008 (Ciclo de vida): este archivo es un insumo

#   describiendo la categoría y el propósito de las dependencias.
# - ISO/IEC 25010:2011 (Calidad): las versiones están fijadas (==) o

# ---------------------------------------------------------------------
# Frameworks Web

fastapi==0.111.0
